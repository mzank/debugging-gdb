# ===================================================
# gdb -x commands_gdb_examplesegfault.txt ./build/examplesegfault
# ===================================================

# Print commands before executing them.
set trace-commands on

# ===================================================
# 1. Identify the segmentation fault.
# ===================================================

run

echo The program stops with a segmentation fault at line 16.\n

# Print the code around the fault.
list 13,17

# Examine the call stack at the fault.
backtrace

# ===================================================
# 2. Inspect the failing statement.
# ===================================================

# Set a breakpoint in main() and at the statement where the fault occurs.
break main
break 16

echo Rerun the program and stop at the source line causing the fault.\n
run

# Start recording program execution for reverse debugging.
record full

# Examine threads.
info threads

# Examine local variables in main().
info locals
echo Local variables have not been initialized yet.\n

# Continue execution to the next breakpoint.
continue

# Examine function arguments of calculate().
info args

# Inspect ptr.
whatis ptr
print ptr
print *ptr

# Execute the next source line without entering called functions.
next
print *ptr

# Continue to the second call of calculate().
continue
print ptr
echo ptr is a null pointer; writing to it causes a segmentation fault.\n
next
echo The segmentation fault occurs here. Remove the line-16 breakpoint before stepping back.\n
# Remove the breakpoint at the fault so reverse stepping can leave the line.
delete 2

echo Step backwards into the previously executed function call.\n
reverse-step

echo Go backwards to the statement that modified the pointer.\n
reverse-next
echo This statement sets *ptr to nullptr.\n

echo Display the currently selected stack frame.\n
frame

echo Display the complete call stack.\n
backtrace

# ===================================================
# 3. Investigate the root cause.
# ===================================================

echo Remove old breakpoints and set one in modify().\n

# Delete the breakpoint in main(); the line-16 breakpoint was deleted earlier.
delete 1

# Print the code.
list 4,11
break 9

# List the breakpoints.
info breakpoints

echo Rerun with a breakpoint in modify().\n
run

# Examine modify().
info args
info locals
print ptr
print *ptr
print **ptr

# Examine CPU registers.
info registers

# Evaluate the if-condition.
print **ptr % 2

continue
print **ptr
print **ptr % 2
echo The first call encounters 4, which is even, so the pointer remains unchanged.\n
echo calculate() then changes the integer to 99.\n
echo The second call encounters 99, which is odd, so modify() sets the local pointer to nullptr.\n

# ===================================================
# 4. Modify the program state to test a hypothesis.
# ===================================================

echo Change the integer value from 99 to the even number 6.\n
set var **ptr=6

# Verify the modified value.
print **ptr
print **ptr % 2

continue
echo The program terminates normally because modify() does not set *ptr to nullptr.\n

# ===================================================
# 5. Clean up.
# ===================================================

delete breakpoints
quit
